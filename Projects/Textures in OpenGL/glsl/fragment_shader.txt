#version 330 core
out vec4 frag_color;

in vec3 color_values;
in vec2 tex;

uniform sampler2D tex_back_data;
uniform sampler2D tex_front_data;

void main()
{
    
    frag_color = mix(texture(tex_back_data, tex) * vec4(color_values,1.0f), texture(tex_front_data, tex), texture(tex_front_data, tex).a * 1.0f);
}
